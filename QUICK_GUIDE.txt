# 查看架构说明
cat README.md

# 查看设计代码
ls src/

# 运行测试
bash scripts/run_ws_test.sh

# 查看测试向量
ls tests/

# 查看综合报告

cat reports/timing_report.txt

# 打开Vivado工程

vivado systolic_array.xpr &
