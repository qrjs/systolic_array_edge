🚀 使用方式
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

【WS架构】
cd ws
cat README.md                    # 查看架构说明
ls src/                          # 查看设计代码
bash scripts/run_ws_test.sh    # 运行测试
python tools/generate_300_tests.py  # 生成测试

【IS架构】
cd is
cat README.md
ls src/
bash scripts/run_is_test.sh
python tools/generate_300_tests.py

【OS架构】
cd os
cat README.md
ls src/
bash scripts/run_os_test.sh
python tools/generate_300_tests.py

【运行所有测试】
bash scripts/run_all_tests.sh
